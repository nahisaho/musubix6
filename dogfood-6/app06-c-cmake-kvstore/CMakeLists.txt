cmake_minimum_required(VERSION 3.16)
project(kvstore C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
include_directories(include)
file(GLOB KV_SRC CONFIGURE_DEPENDS src/*.c)
if(KV_SRC)
  add_library(kv STATIC ${KV_SRC})
  target_compile_options(kv PRIVATE -Wall -Wextra -Werror)
endif()
file(GLOB KV_TESTS CONFIGURE_DEPENDS tests/test_*.c)
foreach(t ${KV_TESTS})
  get_filename_component(tn ${t} NAME_WE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${t})
  add_executable(${tn} ${t})
  if(KV_SRC)
    target_link_libraries(${tn} kv)
  endif()
  file(STRINGS ${t} lines REGEX "^static int (test_[a-z0-9_]+)\\(void\\)")
  foreach(l ${lines})
    string(REGEX REPLACE "^static int (test_[a-z0-9_]+)\\(void\\).*" "\\1" name "${l}")
    add_test(NAME ${name} COMMAND ${tn} ${name})
  endforeach()
endforeach()
