cmake_minimum_required(VERSION 3.16)
project(lsm CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

file(GLOB LSM_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/test_*.cpp)
add_executable(lsm_tests tests/main.cpp ${LSM_TEST_SOURCES})
target_include_directories(lsm_tests PRIVATE include tests)
target_compile_options(lsm_tests PRIVATE -Wall -Wextra)

# one ctest entry per TEST(...) so `ctest -R test_x_001` selects exactly one
foreach(src ${LSM_TEST_SOURCES})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${src})
  file(STRINGS ${src} lines REGEX "^TEST\\(test_")
  foreach(l ${lines})
    string(REGEX REPLACE "^TEST\\((test_[a-z0-9_]+)\\).*" "\\1" name "${l}")
    add_test(NAME ${name} COMMAND lsm_tests ${name})
  endforeach()
endforeach()
