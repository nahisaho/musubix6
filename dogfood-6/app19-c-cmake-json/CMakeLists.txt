cmake_minimum_required(VERSION 3.16)
project(musubix_json C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
enable_testing()

file(GLOB LIB_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.c)
add_library(mjson STATIC ${LIB_SRC})
target_include_directories(mjson PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_link_libraries(mjson PUBLIC m)

file(GLOB TEST_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/t_*.c)
foreach(src ${TEST_SRC})
  get_filename_component(exe ${src} NAME_WE)
  add_executable(${exe} ${src})
  target_link_libraries(${exe} PRIVATE mjson)
  # each "@id TEST-X-nnn" annotation becomes one ctest case test_x_nnn
  file(STRINGS ${src} ids REGEX "@id TEST-")
  foreach(line ${ids})
    string(REGEX MATCH "TEST-[A-Z]+-[0-9]+" id "${line}")
    string(TOLOWER "${id}" lid)
    string(REPLACE "-" "_" lid "${lid}")
    add_test(NAME ${lid} COMMAND ${exe} ${lid})
  endforeach()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${src})
endforeach()
